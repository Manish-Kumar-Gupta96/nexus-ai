import nexus

client = nexus.NexusAI(api_key="nx_live_YOUR_API_KEY")

response = client.chat.completions.create(
    model="nexus-4-turbo",
    messages=[
        {"role": "system", "content": "You are a backend architect."},
        {"role": "user", "content": "Build an event bus in FastAPI with Redis Pub/Sub."}
    ],
    stream=True
)

for chunk in response:
    print(chunk.choices[0].delta.content or "", end="")
import { NexusAI } from '@nexus-ai/sdk';

const client = new NexusAI({ apiKey: process.env.NEXUS_API_KEY });

async function run() {
  const stream = await client.chat.completions.create({
    model: 'nexus-4-turbo',
    messages: [{ role: 'user', content: 'Generate a Next.js 15 Server Action handler' }],
    stream: true,
  });

  for await (const chunk of stream) {
    process.stdout.write(chunk.choices[0]?.delta?.content || '');
  }
}

run();
package main

import (
    "context"
    "fmt"
    "github.com/nexus-ai/nexus-go"
)

func main() {
    client := nexus.NewClient("nx_live_YOUR_API_KEY")
    resp, err := client.CreateChatCompletion(context.Background(), nexus.ChatRequest{
        Model: "nexus-4-turbo",
        Messages: []nexus.Message{{Role: "user", Content: "Explain Goroutines and Mutex locks"}},
    })
    if err != nil {
        panic(err)
    }
    fmt.Println(resp.Choices[0].Message.Content)
}

🛡️ Error Codes & Troubleshooting

Nexus uses standard HTTP response status codes to communicate status outcomes.

HTTP Status Error Type Meaning & Resolution
200 OK Success Request processed and streaming payload returned.
401 Unauthorized Invalid API Key Check Bearer token in the Authorization header.
429 Rate Limit Quota Exceeded Rate limit hit. Implement exponential backoff retry.
500 Server Error Internal Exception Something went wrong on our GPU cluster. Automatic failover will re-route.
